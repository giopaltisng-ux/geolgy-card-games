    <script>
        const cardsData = [
            { id: 1, name: "Batugamping Pesawaran", location: "Pesawaran", ucs: 65, density: 2.55, economic: 85, type: "Sedimen Karbonat" },
            { id: 2, name: "Ignimbit Tanggamus", location: "Tanggamus", ucs: 28, density: 1.95, economic: 60, type: "Vulkanik Pyroklastik" },
            { id: 3, name: "Basalt Sukadana", location: "Lampung Timur", ucs: 140, density: 2.85, economic: 75, type: "Beku Mafik" },
            { id: 4, name: "Lempung Ekspansif", location: "Lampung Selatan", ucs: 8, density: 1.65, economic: 40, type: "Sedimen Pelapukan" },
            { id: 5, name: "Granit Sekampung", location: "Lampung Timur", ucs: 160, density: 2.70, economic: 90, type: "Beku Plutonik" }
        ];

        let selectedPlayerCard = null;
        let selectedAttr = null;
        let playerScore = 0;
        let cpuScore = 0;

        function renderDeck() {
            const container = document.getElementById('player-deck');
            container.innerHTML = '';
            cardsData.forEach(card => {
                const cardEl = document.createElement('div');
                cardEl.className = `bg-slate-800 p-3 rounded-xl border border-slate-700 cursor-pointer hover:border-blue-500 transition text-left`;
                cardEl.onclick = () => pickCard(card, cardEl);
                cardEl.innerHTML = `
                    <div class="text-[10px] text-amber-400 font-semibold">${card.type}</div>
                    <div class="text-xs font-bold my-1 text-white">${card.name}</div>
                    <div class="text-[10px] text-slate-400">${card.location}</div>
                `;
                container.appendChild(cardEl);
            });
        }

        function pickCard(card, element) {
            document.querySelectorAll('#player-deck > div').forEach(el => el.classList.remove('border-blue-500', 'bg-slate-700'));
            element.classList.add('border-blue-500', 'bg-slate-700');
            selectedPlayerCard = card;

            document.getElementById('player-card-slot').innerHTML = renderCardUI(card);
            checkReady();
        }

        function selectAttr(attr) {
            selectedAttr = attr;
            document.querySelectorAll('.attr-btn').forEach(btn => {
                btn.classList.remove('bg-amber-600');
                if(btn.dataset.attr === attr) btn.classList.add('bg-amber-600');
            });
            checkReady();
        }

        function checkReady() {
            document.getElementById('battle-btn').disabled = !(selectedPlayerCard && selectedAttr);
        }

        function renderCardUI(card) {
            return `
                <div class="w-full text-left bg-slate-900 p-3 rounded-lg border border-slate-700">
                    <div class="text-xs text-amber-400 font-bold">${card.name}</div>
                    <div class="text-[10px] text-slate-400 mb-2">${card.location}</div>
                    <div class="text-[11px] space-y-1">
                        <div class="flex justify-between"><span>UCS:</span> <b class="text-blue-300">${card.ucs} MPa</b></div>
                        <div class="flex justify-between"><span>Density:</span> <b class="text-blue-300">${card.density} g/cm³</b></div>
                        <div class="flex justify-between"><span>Nilai Ekonomi:</span> <b class="text-blue-300">${card.economic}/100</b></div>
                    </div>
                </div>
            `;
        }

        function playRound() {
            const randomCpuCard = cardsData[Math.floor(Math.random() * cardsData.length)];
            document.getElementById('cpu-card-slot').innerHTML = renderCardUI(randomCpuCard);

            const pValue = selectedPlayerCard[selectedAttr];
            const cValue = randomCpuCard[selectedAttr];

            const resultBox = document.getElementById('result-box');
            resultBox.classList.remove('hidden', 'bg-green-600', 'bg-red-600', 'bg-slate-600');

            if (pValue > cValue) {
                playerScore++;
                resultBox.classList.add('bg-green-600');
                resultBox.innerText = `Kamu Menang! (${pValue} vs ${cValue})`;
            } else if (pValue < cValue) {
                cpuScore++;
                resultBox.classList.add('bg-red-600');
                resultBox.innerText = `Komputer Menang! (${pValue} vs ${cValue})`;
            } else {
                resultBox.classList.add('bg-slate-600');
                resultBox.innerText = `Seri! Nilai Sama (${pValue})`;
            }

            document.getElementById('player-score').innerText = playerScore;
            document.getElementById('cpu-score').innerText = cpuScore;
        }

        renderDeck();
    </script>
